const express = require('express'); const session = require('express-session'); const multer = require('multer'); const admin = require('firebase-admin'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; const ADMIN_PASSWORD = "Admin@2026"; // ========================================== // 1. إعداد Firebase // ========================================== try { // يقرأ ملف المفاتيح من متغير البيئة أو من ملف محلي const serviceAccount = process.env.FIREBASE_KEY ? JSON.parse(process.env.FIREBASE_KEY) : require('./serviceAccount.json'); admin.initializeApp({ credential: admin.credential.cert(serviceAccount), projectId: "black-9deb2" }); console.log("✅ Firebase Connected"); } catch (error) { console.error("❌ Firebase Error: تأكد من إعداد serviceAccount.json أو متغير البيئة"); } const db = admin.firestore(); // ========================================== // 2. إعدادات السيرفر // ========================================== app.use(express.urlencoded({ extended: true })); app.use(session({ secret: 'super-secret-key', resave: false, saveUninitialized: true, cookie: { secure: false } })); const upload = multer({ storage: multer.memoryStorage() }); // ========================================== // 3. دالة التصميم (HTML) // ========================================== const layout = (content) => ` بلدنا للمدفوعات

بلدنا

للمدفوعات الرقمية

الإدارة
${content}
`; // ========================================== // 4. موقع العملاء (الواجهة) // ========================================== app.get('/', (req, res) => { const content = `

طلب إيداع أو سحب

${req.query.success ? '
تم إرسال الطلب بنجاح!
' : ''} ${req.query.error ? '
حدث خطأ، حاول مرة أخرى.
' : ''}

بيانات السحب

`; res.send(layout(content)); }); // استقبال الطلب وحفظه app.post('/submit', upload.single('receipt'), async (req, res) => { try { const { full_name, account_number, type, amount, sender_name, beneficiary_name, beneficiary_account, bank_ref } = req.body; let receiptUrl = ""; if (req.file) { const bucket = admin.storage().bucket("black-9deb2.appspot.com"); const blob = bucket.file(`receipts/${Date.now()}_${req.file.originalname}`); const blobStream = blob.createWriteStream({ resumable: false }); blobStream.end(req.file.buffer); await new Promise(resolve => blobStream.on('finish', resolve)); await blob.makePublic(); receiptUrl = `https://storage.googleapis.com/${bucket.name}/${blob.name}`; } // البحث عن المستخدم أو إنشائه let userId; const usersRef = db.collection('users'); const snapshot = await usersRef.where('account_number', '==', account_number).get(); if (!snapshot.empty) { userId = snapshot.docs[0].id; } else { const newUser = await usersRef.add({ username: full_name, full_name, account_number, balance: 0, status: 'active', created_at: new Date().toISOString() }); userId = newUser.id; } // إضافة الطلب await db.collection('transactions').add({ user_id: userId, type, amount: parseFloat(amount), sender_name: sender_name || "", beneficiary_name: beneficiary_name || "", beneficiary_account: beneficiary_account || "", bank_ref: bank_ref || "", receipt_url: receiptUrl, status: 'pending', created_at: new Date().toISOString() }); res.redirect('/?success=1'); } catch (error) { console.error(error); res.redirect('/?error=1'); } }); // ========================================== // 5. لوحة تحكم المدير // ========================================== app.get('/admin', (req, res) => { if (!req.session.admin) { return res.send(layout(`

دخول الإدارة

${req.query.error ? '

كلمة المرور خاطئة

' : ''}
`)); } res.redirect('/admin/dashboard'); }); app.post('/admin/login', (req, res) => { if (req.body.password === ADMIN_PASSWORD) { req.session.admin = true; res.redirect('/admin/dashboard'); } else { res.redirect('/admin?error=1'); } }); app.get('/admin/logout', (req, res) => { req.session.destroy(); res.redirect('/admin'); }); app.get('/admin/dashboard', async (req, res) => { if (!req.session.admin) return res.redirect('/admin'); try { const txSnap = await db.collection('transactions').orderBy('created_at', 'desc').get(); const usersSnap = await db.collection('users').get(); let users = {}; usersSnap.forEach(d => users[d.id] = d.data()); let txHtml = ''; let pendingCount = 0; txSnap.forEach(doc => { const tx = doc.data(); if (tx.status === 'pending') pendingCount++; const u = users[tx.user_id] || {}; const isDeposit = tx.type === 'deposit'; txHtml += `
${isDeposit ? 'إيداع' : 'سحب'} #${doc.id.substring(0,6)} ${tx.status}
العميل: ${u.full_name || '—'}
المبلغ: ${parseFloat(tx.amount).toLocaleString()} SDG
الهاتف: ${u.account_number || '—'}
${tx.receipt_url ? ` عرض الإشعار` : ''}
${tx.status === 'pending' ? `
موافقة رفض
` : ''}
`; }); res.send(layout(`

لوحة الإدارة (قيد الانتظار: ${pendingCount})

خروج
${txHtml || '

لا توجد طلبات حالياً

'}
`)); } catch (e) { res.send("Error: " + e.message); } }); app.get('/admin/action', async (req, res) => { if (!req.session.admin) return res.redirect('/admin'); const { action, id } = req.query; try { const txRef = db.collection('transactions').doc(id); const txDoc = await txRef.get(); if (txDoc.exists && txDoc.data().status === 'pending') { const tx = txDoc.data(); if (action === 'approve') { if (tx.user_id) { const userRef = db.collection('users').doc(tx.user_id); const userDoc = await userRef.get(); if (userDoc.exists) { const current = parseFloat(userDoc.data().balance || 0); const amount = parseFloat(tx.amount || 0); const newBal = tx.type === 'deposit' ? current + amount : current - amount; await userRef.update({ balance: newBal, account_balance: newBal }); } } await txRef.update({ status: 'approved' }); } else { await txRef.update({ status: 'rejected' }); } } res.redirect('/admin/dashboard'); } catch (e) { res.send(e.message); } }); app.listen(PORT, () => console.log(`🚀 Server on port ${PORT}`));
